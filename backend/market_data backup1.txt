async def get_crypto_snapshot(symbol: str, timeframe: str) -> dict[str, Any] | None:
    sym_upper = symbol.upper()
    
    # Penanganan khusus aset USDT agar selalu muncul dengan nilai stabil 1.0
    if sym_upper == "USDT":
        now_ts = int(time.time())
        candles = [
            {
                "time": now_ts - (i * 3600),
                "open": 1.0,
                "high": 1.0005,
                "low": 0.9995,
                "close": 1.0,
                "volume": 1000000.0,
            }
            for i in range(100, 0, -1)
        ]
        ticker = {
            "price": 1.0,
            "change_pct_24h": 0.0,
            "high_24h": 1.0005,
            "low_24h": 0.9995,
            "volume_24h": 10000000.0,
        }
        return {"symbol": "USDT", "market": "crypto", "ticker": ticker, "candles": candles}

    sym = CRYPTO_BINANCE_MAP.get(sym_upper, f"{sym_upper}USDT")
    url = f"https://api.binance.com/api/v3/klines?symbol={sym}&interval={timeframe}&limit=100"
    async with httpx.AsyncClient(timeout=10.0) as client:
        try:
            resp = await client.get(url)
            if resp.status_code != 200:
                return None
            data = resp.json()
            candles = []
            for row in data:
                candles.append({
                    "time": int(row[0]) // 1000,
                    "open": float(row[1]),
                    "high": float(row[2]),
                    "low": float(row[3]),
                    "close": float(row[4]),
                    "volume": float(row[5]),
                })
            if not candles:
                return None
            last = candles[-1]
            ticker = {
                "price": last["close"],
                "change_pct_24h": 0.0,
                "high_24h": max(c["high"] for c in candles[-24:]),
                "low_24h": min(c["low"] for c in candles[-24:]),
                "volume_24h": sum(c["volume"] for c in candles[-24:]),
            }
            return {"symbol": sym_upper, "market": "crypto", "ticker": ticker, "candles": candles}
        except Exception:
            return None

async def get_stock_snapshot(symbol: str, timeframe: str, market: str = "stock_us") -> dict[str, Any] | None:
    sym = symbol.upper()
    if market == "stock_id" and not sym.endswith(".JK"):
        sym += ".JK"
    elif market == "gold_silver" and sym in GOLD_SILVER_YAHOO_MAP:
        sym = GOLD_SILVER_YAHOO_MAP[sym]
    elif market == "commodity" and sym in COMMODITY_YAHOO_MAP:
        sym = COMMODITY_YAHOO_MAP[sym]
    elif market == "forex" and sym in FOREX_YAHOO_MAP:
        sym = FOREX_YAHOO_MAP[sym]

    url = f"https://query1.finance.yahoo.com/v8/finance/chart/{sym}?interval=1h&range=5d"
    headers = {"User-Agent": "Mozilla/5.0"}
    async with httpx.AsyncClient(timeout=10.0, headers=headers) as client:
        try:
            resp = await client.get(url)
            if resp.status_code != 200:
                return None
            res = resp.json()
            result = res.get("chart", {}).get("result")
            if not result:
                return None
            meta = result[0].get("meta", {})
            timestamps = result[0].get("timestamp", [])
            quote = result[0].get("indicators", {}).get("quote", [{}])[0]
            
            closes = quote.get("close", [])
            opens = quote.get("open", [])
            highs = quote.get("high", [])
            lows = quote.get("low", [])
            volumes = quote.get("volume", [])

            candles = []
            for i in range(len(timestamps)):
                if i < len(closes) and closes[i] is not None:
                    candles.append({
                        "time": int(timestamps[i]),
                        "open": float(opens[i] if opens[i] is not None else closes[i]),
                        "high": float(highs[i] if highs[i] is not None else closes[i]),
                        "low": float(lows[i] if lows[i] is not None else closes[i]),
                        "close": float(closes[i]),
                        "volume": float(volumes[i] if volumes[i] is not None else 0),
                    })
            if not candles:
                return None
            last = candles[-1]
            current_price = meta.get("regularMarketPrice", last["close"])
            prev_close = meta.get("chartPreviousClose", last["close"])
            change_pct = ((current_price - prev_close) / prev_close) * 100 if prev_close else 0.0

            ticker = {
                "price": current_price,
                "change_pct_24h": round(change_pct, 2),
                "high_24h": max((c["high"] for c in candles[-24:]), default=current_price),
                "low_24h": min((c["low"] for c in candles[-24:]), default=current_price),
                "volume_24h": sum(c["volume"] for c in candles[-24:]),
            }
            return {"symbol": symbol.upper(), "market": market, "ticker": ticker, "candles": candles}
        except Exception:
            return None

async def get_batch(market: str, symbols: list[str], timeframe: str) -> list[dict[str, Any]]:
    tasks = [
        get_crypto_snapshot(s, timeframe) if market == "crypto" else get_stock_snapshot(s, timeframe, market=market)
        for s in symbols
    ]
    res = await asyncio.gather(*tasks, return_exceptions=True)
    return [r for r in res if isinstance(r, dict)]
